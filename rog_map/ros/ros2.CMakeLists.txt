cmake_minimum_required(VERSION 3.5)
project(rog_map)

set(CMAKE_CXX_STANDARD 17)
#set(CMAKE_BUILD_TYPE "Debug")
set(CMAKE_BUILD_TYPE "Release")
# [px4_oa] -ffp-contract=off: no fused multiply-add, so the CPU path and the CUDA kernels
# (-fmad=false) round the same way and their maps can be compared exactly.
set(CMAKE_CXX_FLAGS "-O3 -Wall -g -fPIC -ffp-contract=off")
add_compile_options(-DQT_NO_VERSION_TAGGING)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Werror=unused-variable>)  # [px4_oa] not for nvcc
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Werror=return-type>)  # [px4_oa] not for nvcc

# reduce warning
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Wno-dev>)  # [px4_oa] not for nvcc
cmake_policy(SET CMP0057 NEW)
cmake_policy(SET CMP0054 NEW)
set(CMAKE_SUPPRESS_DEVELOPER_WARNINGS 1 CACHE BOOL "Suppress developer warnings")

add_definitions(-DROOT_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/\")

# Define the voxelize and raycasting method
add_definitions(-DORIGIN_AT_CORNER)

string(TOUPPER $ENV{ROS_DISTRO} ROS_VERSION)
message(STATUS "ROS version: ${ROS_VERSION}")

message("Using ROS2")
add_definitions(-DUSE_ROS2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(PCL REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)

set(includes
        include
        ${Eigen_INCLUDE_DIRS}
        #            ${OpenCV_INCLUDE_DIRS}
        ${PCL_INCLUDE_DIRS}
)

set(third_party_libs
        #            ${OpenCV_LIBRARIES}
        ${PCL_LIBRARIES}
        yaml-cpp
)

set(ros_libs
        rclcpp
        std_msgs
        nav_msgs
        geometry_msgs
        visualization_msgs
        pcl_conversions
        tf2_ros
)

include_directories(
        ${includes}
)

file(GLOB_RECURSE srcs CONFIGURE_DEPENDS src/*.cpp include/*.h include/*.hpp include/*.cpp)

# [px4_oa] CUDA backend for the probabilistic update (docs/cuda_rogmap_plan.md). Built when a
# CUDA compiler is found (PATH, CUDACXX or /usr/local/cuda); selected at run time with
# rog_map/raycasting/backend: cuda. -DROGMAP_CUDA=OFF builds without it.
option(ROGMAP_CUDA "Build the CUDA raycasting backend if a CUDA compiler is found" ON)
if (ROGMAP_CUDA)
    if (NOT CMAKE_CUDA_COMPILER AND NOT DEFINED ENV{CUDACXX} AND EXISTS /usr/local/cuda/bin/nvcc)
        set(CMAKE_CUDA_COMPILER /usr/local/cuda/bin/nvcc)
    endif ()
    include(CheckLanguage)
    check_language(CUDA)
    if (CMAKE_CUDA_COMPILER)
        enable_language(CUDA)
        find_package(CUDAToolkit REQUIRED)
        set(CMAKE_CUDA_STANDARD 17)
        set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} -O3 -fmad=false")
        list(APPEND srcs src/cuda/raycast_backend.cu)
        message(STATUS "rog_map: CUDA backend on (${CMAKE_CUDA_COMPILER})")
    else ()
        message(STATUS "rog_map: no CUDA compiler found, CPU backend only")
        set(ROGMAP_CUDA OFF)
    endif ()
endif ()

add_library(rog_map STATIC ${srcs})
if (ROGMAP_CUDA)
    target_compile_definitions(rog_map PUBLIC ROGMAP_CUDA)
    target_link_libraries(rog_map PUBLIC ${CUDA_cudart_LIBRARY})
    list(APPEND third_party_libs ${CUDA_cudart_LIBRARY})
endif ()

target_include_directories(rog_map PUBLIC include)
target_link_libraries(rog_map PUBLIC
        ${THIRD_PARTY})


ament_target_dependencies(${PROJECT_NAME} PUBLIC
        ${ros_libs}
)
target_link_libraries(${PROJECT_NAME} PUBLIC
        ${third_party_libs}
)


# [px4_oa] Nsight Systems ranges (NVTX is header-only and ships with the CUDA toolkit)
find_path(NVTX_INCLUDE_DIR nvtx3/nvToolsExt.h PATHS /usr/local/cuda/include)
if (NVTX_INCLUDE_DIR)
    message(STATUS "rog_map: NVTX ranges on (${NVTX_INCLUDE_DIR})")
    target_compile_definitions(${PROJECT_NAME} PUBLIC ROGMAP_NVTX)
    target_include_directories(${PROJECT_NAME} PUBLIC ${NVTX_INCLUDE_DIR})
    target_link_libraries(${PROJECT_NAME} PUBLIC ${CMAKE_DL_LIBS})
endif ()

# [px4_oa] replay benchmark, see docs/cuda_rogmap_plan.md
add_executable(rogmap_bench apps/rogmap_bench.cpp)
target_link_libraries(rogmap_bench rog_map)
install(TARGETS rogmap_bench RUNTIME DESTINATION lib/${PROJECT_NAME})

install(TARGETS
        ${PROJECT_NAME}
        ARCHIVE DESTINATION lib
        LIBRARY DESTINATION lib
        RUNTIME DESTINATION bin
)

ament_export_libraries(${PROJECT_NAME}
        ${third_party_libs}
)
ament_export_include_directories(${includes})
install(DIRECTORY include/
        DESTINATION include/
)

ament_package()
